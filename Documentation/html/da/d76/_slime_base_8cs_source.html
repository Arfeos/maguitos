279 protected virtual IEnumerator DissolveAfterDelay()
280 {
281 yield return new WaitForSeconds(delayBeforeStart);
282
283 Renderer[] renderers = GetComponentsInChildren<Renderer>();
284
285 // Cambiar shader en todos los materiales
286 foreach (Renderer r in renderers)
287 {
288 foreach (Material mat in r.materials)
289 {
290 mat.shader = Shader.Find("Custom/Dissolve");
291 mat.SetTexture("_DissolveTex", dissolveTexture);
292 mat.SetColor("_DissolveColor", dissolveColor);
293 }
294 }
295
296 // Animar la disolución
297 while (dissolveProgress < 1f)
298 {
299 dissolveProgress += Time.deltaTime / dissolveTime;
300
301 foreach (Renderer r in renderers)
302 foreach (Material mat in r.materials)
303 mat.SetFloat("_DissolveThreshold", dissolveProgress);
304
305 yield return null;
306 }
307
308 Destroy(gameObject);
309 }
